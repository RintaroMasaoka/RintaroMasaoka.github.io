\section*{[1] 1粒子状態のエネルギー}
図の $\psi_S$ は節を持たない対称状態、$\psi_A$ は節を1つ持つ反対称状態である。一次元束縛状態では節の少ない状態ほどエネルギーが低いので
\begin{equation}
 E_S=-J,\qquad E_A=+J.
\end{equation}

\section*{[2] 局在基底}
規格直交する $\psi_S,\psi_A$ より
\begin{equation}
 \langle\psi_L|\psi_R\rangle
 =\frac12(\langle\psi_S|\psi_S\rangle-\langle\psi_A|\psi_A\rangle)=0.
\end{equation}
また $H\psi_S=-J\psi_S$, $H\psi_A=J\psi_A$ だから、$\{\psi_L,\psi_R\}$ 基底の行列要素は
\begin{equation}
 \boxed{H_1=\begin{pmatrix}0&-J\\-J&0\end{pmatrix}.}
\end{equation}

\section*{[3] 1粒子のトンネル確率}
左井戸状態 $|L\rangle$ からの時間発展は
\begin{equation}
 e^{-iH_1t/\hbar}|L\rangle
 =\cos\!\left(\frac{Jt}{\hbar}\right)|L\rangle
 +i\sin\!\left(\frac{Jt}{\hbar}\right)|R\rangle.
\end{equation}
従って右井戸にある確率は
\begin{equation}
 \boxed{P_R(t)=\sin^2\!\left(\frac{Jt}{\hbar}\right).}
\end{equation}

\section*{[4] 相互作用のない2粒子系}
\subsection*{[4.1] ハミルトニアン}
対称化された規格化状態は
\begin{equation}
 |0\rangle=|LL\rangle,\qquad
 |1\rangle=\frac{|LR\rangle+|RL\rangle}{\sqrt2},\qquad
 |2\rangle=|RR\rangle.
\end{equation}
一体トンネル項 $-J(|L\rangle\langle R|+|R\rangle\langle L|)$ を各粒子に作用させると
\begin{equation}
 H|0\rangle=-\sqrt2J|1\rangle,\quad
 H|1\rangle=-\sqrt2J(|0\rangle+|2\rangle),\quad
 H|2\rangle=-\sqrt2J|1\rangle.
\end{equation}
従って
\begin{equation}
 \boxed{H_0=\begin{pmatrix}
 0&-\sqrt2J&0\\-\sqrt2J&0&-\sqrt2J\\0&-\sqrt2J&0
 \end{pmatrix}.}
\end{equation}
\subsection*{[4.2] 固有エネルギー}
反対称な端状態 $(|0\rangle-|2\rangle)/\sqrt2$ の固有値は0。対称な端状態 $(|0\rangle+|2\rangle)/\sqrt2$ と $|1\rangle$ に対して、上の行列の制限は $\left(\begin{smallmatrix}0&-2J\\-2J&0\end{smallmatrix}\right)$ となる。この二次元ブロックの固有値は $\pm2J$ なので
\begin{equation}
 \boxed{E=-2J,\ 0,\ 2J.}
\end{equation}
\subsection*{[4.3] 粒子数確率}
初期状態 $|LL\rangle$ は、独立に左右へ移る2粒子の積状態である。$x=Jt/\hbar$ とすると
\begin{align}
 P_0(t)&=\cos^4x,\\
 P_1(t)&=2\sin^2x\cos^2x=\frac12\sin^2(2x),\\
 P_2(t)&=\sin^4x.
\end{align}
各時刻で和は1。原図の選択肢では図2(a)に対応し、点線が $P_0$、実線が $P_1$、一点鎖線が $P_2$ を表す。確率の共通周期は
\begin{equation}
 \boxed{T=\frac{\pi\hbar}{J}}.
\end{equation}
一周期の概形は次のとおりである。
\begin{center}

\section*{[5] 強い引力相互作用}
\subsection*{[5.1] ハミルトニアン}
同一井戸に2粒子がある状態のエネルギーを $-A$ だけ下げると、
\begin{equation}
 \boxed{H_{-A}=\begin{pmatrix}
 -A&-\sqrt2J&0\\-\sqrt2J&0&-\sqrt2J\\0&-\sqrt2J&-A
 \end{pmatrix}.}
\end{equation}
\subsection*{[5.2] 固有エネルギー}
反対称端状態のエネルギーは $-A$。対称端状態と $|1\rangle$ の部分行列は
$\begin{pmatrix}-A&-2J\\-2J&0\end{pmatrix}$ である。従って
\begin{equation}
 E=-A,\qquad E_\pm=\frac{-A\pm\sqrt{A^2+16J^2}}2.
\end{equation}
$A\gg J$ で展開すると
\begin{equation}
 \boxed{E_-=-A-\frac{4J^2}{A}+O(J^4/A^3),\quad
 E_a=-A,\quad E_+=\frac{4J^2}{A}+O(J^4/A^3).}
\end{equation}
\subsection*{[5.3] 引力の場合の粒子数確率}
有限の $A$ に対する確率を求める。同一井戸状態の相互作用エネルギーを一般の符号付き $u$ とし、そのハミルトニアンを $H(u)$ と書く。引力の場合は $u=-A$ である。端状態の対称・反対称な組合せを
\begin{equation}
 |s\rangle=\frac{|0\rangle+|2\rangle}{\sqrt2},\qquad
 |a\rangle=\frac{|0\rangle-|2\rangle}{\sqrt2}
\end{equation}
と置く。[5.2]と同じ分解により、基底 $(|s\rangle,|1\rangle)$ での部分行列は
\begin{equation}
 H_s(u)=\begin{pmatrix}u&-2J\\-2J&0\end{pmatrix},
 \qquad H(u)|a\rangle=u|a\rangle
\end{equation}
である。準位差を $\Delta=\sqrt{u^2+16J^2}$ とすると、2次元の単位行列 $\mathbf I$ に対して
\begin{equation}
 \left(H_s(u)-\frac u2\mathbf I\right)^2
 =\frac{\Delta^2}{4}\mathbf I.
\end{equation}
従って行列指数の偶数べきと奇数べきをまとめれば時間発展を求められる。この小節では $x=\Delta t/(2\hbar)$、$y=ut/(2\hbar)$ と置く。初期状態 $|0\rangle=(|s\rangle+|a\rangle)/\sqrt2$ から
\begin{align}
 |\Psi(t)\rangle
 &=\frac{e^{-iy}}{\sqrt2}
   \left[\left(\cos x-i\frac u\Delta\sin x\right)|s\rangle
          +i\frac{4J}{\Delta}\sin x\,|1\rangle\right]\\
 &\quad+\frac{e^{-2iy}}{\sqrt2}|a\rangle.
\end{align}
各占有状態へ射影して振幅の絶対値二乗を取ると
\begin{align}
 P_0(t)&=\frac14\left|e^{-iy}\left(\cos x-i\frac u\Delta\sin x\right)+e^{-2iy}\right|^2,\\
 P_1(t)&=\frac{8J^2}{\Delta^2}\sin^2x,\\
 P_2(t)&=\frac14\left|e^{-iy}\left(\cos x-i\frac u\Delta\sin x\right)-e^{-2iy}\right|^2.
\end{align}
従って有限 $A$ では中間状態 $|1\rangle$ も一般に占有される。$A/J\gg1$ ではその振幅は小さく、主要な時間変化は二つの同一井戸状態間の遅いペア移動となる。初期状態 $|0\rangle$ は対称状態 $|s\rangle$ と反対称状態 $|a\rangle$ を等重みで含む。$|s\rangle$ を主成分とする低エネルギー固有状態と $|a\rangle$ のエネルギー差は $4J^2/A$ であり、その相対位相から
\begin{equation}
 P_0(t)\simeq\cos^2\!\left(\frac{2J^2t}{A\hbar}\right),\qquad
 P_2(t)\simeq\sin^2\!\left(\frac{2J^2t}{A\hbar}\right),\qquad P_1\ll1.
\end{equation}
ペア移動の周期は
\begin{equation}
 \boxed{T_A=\frac{2\pi\hbar}{4J^2/A}=\frac{\pi\hbar A}{2J^2}}.
\end{equation}
図2(b)に対応する概形は次のとおりである。点線が $P_0$、実線が $P_2$、一点鎖線が小さい $P_1$ である。
\begin{center}
\begin{tikzpicture}[x=1.15cm,y=1.25cm,>=Stealth]
 \draw[->] (0,0)--(4.3,0) node[right] {$t/T_A$};
 \draw[->] (0,0)--(0,1.15) node[above] {$P_n$};
 \draw[dotted,thick,domain=0:2.5,samples=80] plot (\x,{cos(deg(pi*\x))^2});
 \draw[thick,domain=0:2.5,samples=80] plot (\x,{sin(deg(pi*\x))^2});
 \draw[dash dot,thick] (0,0.015)--(2.5,0.015);
 \draw (1,0)--(1,-0.04) node[below] {1};
 \node[anchor=west] at (2.9,0.86) {$P_0$ dotted};

 \node[anchor=west] at (2.9,0.25) {$P_1$ dash-dot};
\end{tikzpicture}
\end{center}
\section*{[6] 強い斥力相互作用}
同じ井戸に2粒子があるときのエネルギーを $B$ だけ上げると
\begin{equation}
 H_{+B}=\begin{pmatrix}
 B&-\sqrt2J&0\\-\sqrt2J&0&-\sqrt2J\\0&-\sqrt2J&B
 \end{pmatrix},\qquad
 E=B,\quad E_\pm=\frac{B\pm\sqrt{B^2+16J^2}}2.
\end{equation}
$B\gg J$ では $E_-= -4J^2/B+O(J^4/B^3)$、$E_a=B$、$E_+=B+4J^2/B+O(J^4/B^3)$。従って近接する端状態間の遅いペア移動も同じ形であり
\begin{equation}
 P_0(t)\simeq\cos^2\!\left(\frac{2J^2t}{B\hbar}\right),\qquad
 P_2(t)\simeq\sin^2\!\left(\frac{2J^2t}{B\hbar}\right),\qquad P_1\ll1.
\end{equation}
原図では同じく図2(b)で、周期は
\begin{equation}
 \boxed{T_B=\frac{\pi\hbar B}{2J^2}}.
\end{equation}
ここでも有限 $B$ では $P_1(t)$ は厳密にはゼロでなく、[5.3]の厳密式に $u=B$ を入れる。

\section*{[7] 引力と斥力の比較}
引力 $u=-A$ と斥力 $u=+B$ のどちらでも、同じ絶対値 $|u|$ なら厳密な確率式は同じである。行列
\begin{equation}
 H(u)=\begin{pmatrix}u&-\sqrt2J&0\\-\sqrt2J&0&-\sqrt2J\\0&-\sqrt2J&u\end{pmatrix}
\end{equation}
に $S=\operatorname{diag}(1,-1,1)$ を使うと $H(-u)=-S H(u)S$。$H(u)$ は実行列で $S|0\rangle=|0\rangle$ だから、時間発展は
\begin{equation}
 e^{-iH(-u)t/\hbar}|0\rangle
 =S\bigl[e^{-iH(u)t/\hbar}|0\rangle\bigr]^*.
\end{equation}
$S$ は各占有振幅に符号 $\pm1$ を掛けるだけなので、複素共役後の絶対値二乗も元と同じである。従って各占有数確率と確率グラフの概形は符号反転で変化しない。違うのは同一井戸の対状態が他状態より低いか高いかというエネルギー準位の順序である。
